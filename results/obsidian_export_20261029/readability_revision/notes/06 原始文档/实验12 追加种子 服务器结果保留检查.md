import json
from pathlib import Path

old = Path('results/exp12_sources_20261014_v2/runs_700454')
new = Path('results/exp12_seed345_20261014/runs_700972')
for root in (old, new):
    marker = json.loads((root / 'complete.json').read_text())
    assert marker['training_runs'] == 6 and not marker['smoke_test'], root

def rows(path):
    return [json.loads(s) for s in path.read_text().splitlines() if s.strip()]

plan = rows(old / 'plan/items_D_reservedC.jsonl')
keys = [(r['category'], r['person']) for r in plan]
assert len(keys) == 500 and len(set(keys)) == 500
folders = [old / 'baseline']
for seed in range(6):
    root = old if seed < 3 else new
    for condition in ('context', 'direct'):
        folder = root / f'{condition}_s{seed}'
        done = json.loads((folder / 'complete.json').read_text())
        assert done['steps'] == 216, folder
        for name in ('adapter/adapter_model.safetensors', 'run_config.json', 'train_examples.jsonl', 'loss.jsonl'):
            f = folder / name
            assert f.is_file() and f.stat().st_size > 0, f
        folders.append(folder)
for folder in folders:
    data = rows(folder / 'eval_D_reservedC.jsonl')
    assert [(r['category'], r['person']) for r in data] == keys, folder
    assert all(all(k in r for k in ('pred', 'q1_pred', 'q2_pred')) for r in data), folder
assert (old / 'baseline/eval_D_reservedC.jsonl').read_bytes() == (new / 'reused_700454/baseline/eval_D_reservedC.jsonl').read_bytes()
print('OK: baseline + 12 trained models; all 13 prediction files contain the same 500 D items.')
print('OK: all 12 adapter files exist and are nonempty (presence check, not weight hashing).')
print('Results retained. Await the new scorer; no files changed or jobs submitted.')
PY_CHECK
